Structural pattern

Flyweight

Flyweight giúp rất nhiều đối tượng nhỏ dùng chung phần dữ liệu nặng và không đổi, thay vì mỗi đối tượng giữ một bản sao riêng. Phần riêng của từng đối tượng vẫn ở nguyên chỗ cũ, còn phần chung chỉ tồn tại đúng một lần trong bộ nhớ.

Bài toán

Hãy hình dung một job dựng báo cáo bán hàng, nạp cùng lúc 100.000 dòng đơn hàng vào bộ nhớ. Dòng nào cũng cần biết sản phẩm của nó, gồm tên, phần mô tả dài và thông tin ảnh. Điều đáng chú ý là 100.000 dòng ấy chỉ thuộc về 300 sản phẩm khác nhau.

Phần dữ liệuVí dụƯớc tínhĐặc điểm
Thông tin sản phẩmtên, mô tả, ảnh2.000 bytegiống hệt nhau ở mọi dòng cùng sản phẩm
Phần riêng của dòngsố lượng, mã đơn64 bytemỗi dòng một khác

Lô nhỏ trong hình là mười hai dòng đầu của báo cáo, thuộc ba sản phẩm. Các con số kích thước là ước tính đơn giản để thấy tỉ lệ, với quy ước 1 KB bằng 1.000 byte.

Khi chưa có Flyweight

Cách viết tự nhiên nhất là để mỗi dòng giữ luôn đối tượng sản phẩm của riêng nó, đọc từ cơ sở dữ liệu ra là tạo mới:

Chạy: python3 before.py

LINE_BYTES = 64       # phần riêng của một dòng: số lượng, mã đơn...

    "SP-001": ("Bình giữ nhiệt", "Thép không gỉ hai lớp, giữ nóng 12 giờ, dung tích 500 ml.", "sp-001.jpg 800x800"),

    def __init__(self, code: str) -> None:
        ProductInfo.made += 1
        self.code = code

@dataclass

    product: ProductInfo  # bản sao riêng của dòng này

    return [OrderLine(ProductInfo(code), quantity) for code, quantity in rows]

Lời giải

Hãy tách dữ liệu của mỗi dòng thành hai phần. Phần nặng và không đổi của sản phẩm, thường gọi là intrinsic state, được gom vào một đối tượng Product bất biến, và đó chính là flyweight. Phần riêng của từng dòng như số lượng, thường gọi là extrinsic state, vẫn nằm trong OrderLine và đi kèm mỗi lần dùng tới sản phẩm.

Lớp ProductCatalog đóng vai factory. Nó giữ một bảng tra theo mã sản phẩm, lần đầu gặp một mã thì tạo đối tượng rồi cất lại, những lần sau trả về đúng đối tượng ấy. Nhờ vậy mỗi sản phẩm chỉ tồn tại một lần, dù có bao nhiêu dòng trỏ tới nó.

"""Flyweight: các dòng đơn hàng dùng chung một đối tượng cho mỗi sản phẩm.

Chạy: python3 flyweight.py
"""
from dataclasses import FrozenInstanceError, dataclass

# Kích thước ước tính, tính bằng byte.
PRODUCT_BYTES = 2000  # tên, mô tả dài và thông tin ảnh của một sản phẩm
LINE_BYTES = 64       # phần riêng của một dòng: số lượng, mã đơn...
REF_BYTES = 8         # một tham chiếu tới sản phẩm dùng chung

CATALOG = {
    "SP-001": ("Bình giữ nhiệt", "Thép không gỉ hai lớp, giữ nóng 12 giờ, dung tích 500 ml.", "sp-001.jpg 800x800"),
    "SP-002": ("Hộp trà", "Trà ô long Bảo Lộc, hộp thiếc 200 g.", "sp-002.jpg 800x800"),
    "SP-003": ("Ly sứ", "Ly sứ trắng men bóng, 350 ml, dùng được trong lò vi sóng.", "sp-003.jpg 800x800"),
}


def details(code: str) -> tuple[str, str, str]:
    """Đọc tên, mô tả và ảnh của một sản phẩm từ cơ sở dữ liệu."""
    return CATALOG.get(code, (f"Sản phẩm {code}", f"Mô tả chi tiết của {code}.", f"{code.lower()}.jpg 800x800"))


# Flyweight: phần nặng và không đổi của sản phẩm. frozen=True để không ai sửa được nó,
# vì sửa một đối tượng dùng chung là sửa luôn mọi dòng đang trỏ tới nó.
@dataclass(frozen=True)
class Product:
    code: str
    name: str
    description: str
    image: str


# Factory: mỗi mã sản phẩm chỉ tạo đúng một đối tượng, những lần sau trả lại đối tượng cũ.
class ProductCatalog:
    def __init__(self) -> None:
        self._made: dict[str, Product] = {}

    def get(self, code: str) -> Product:
        product = self._made.get(code)
        if product is None:
            product = Product(code, *details(code))
            self._made[code] = product
        return product

    def __len__(self) -> int:
        return len(self._made)


# Phần riêng của từng dòng: số lượng, cùng một tham chiếu tới sản phẩm dùng chung.
@dataclass
class OrderLine:
    product: Product
    quantity: int

    def label(self) -> str:
        return f"{self.product.name} x{self.quantity}"


def group(n: int) -> str:
    return f"{n:,}".replace(",", ".")


def size(nbytes: int) -> str:
    value, unit = (nbytes / 1e6, "MB") if nbytes >= 1e6 else (nbytes / 1000, "KB")
    return f"{value:.1f} {unit}".replace(".", ",")


def summary(title: str, rows: list[tuple[str, int]]) -> list[OrderLine]:
    catalog = ProductCatalog()
    lines = [OrderLine(catalog.get(code), quantity) for code, quantity in rows]
    memory = len(catalog) * PRODUCT_BYTES + len(lines) * (LINE_BYTES + REF_BYTES)
    print(title)
    print(f"  đối tượng sản phẩm: {group(len(catalog))}")
    print(f"  bộ nhớ ước tính: {size(memory)}")
    return lines

// Flyweight: các dòng đơn hàng dùng chung một đối tượng cho mỗi sản phẩm.
// Chạy: node flyweight.ts   (Node 22.18 trở lên chạy thẳng TypeScript)
"use strict";

// Kích thước ước tính, tính bằng byte.
const PRODUCT_BYTES = 2000; // tên, mô tả dài và thông tin ảnh của một sản phẩm
const LINE_BYTES = 64; // phần riêng của một dòng: số lượng, mã đơn...
const REF_BYTES = 8; // một tham chiếu tới sản phẩm dùng chung

const CATALOG: Record<string, [string, string, string]> = {
  "SP-001": ["Bình giữ nhiệt", "Thép không gỉ hai lớp, giữ nóng 12 giờ, dung tích 500 ml.", "sp-001.jpg 800x800"],
  "SP-002": ["Hộp trà", "Trà ô long Bảo Lộc, hộp thiếc 200 g.", "sp-002.jpg 800x800"],
  "SP-003": ["Ly sứ", "Ly sứ trắng men bóng, 350 ml, dùng được trong lò vi sóng.", "sp-003.jpg 800x800"],
};

// Đọc tên, mô tả và ảnh của một sản phẩm từ cơ sở dữ liệu.
function details(code: string): [string, string, string] {
  return CATALOG[code] ?? [`Sản phẩm ${code}`, `Mô tả chi tiết của ${code}.`, `${code.toLowerCase()}.jpg 800x800`];
}

// Flyweight: phần nặng và không đổi của sản phẩm. Object.freeze để không ai sửa được nó,
// vì sửa một đối tượng dùng chung là sửa luôn mọi dòng đang trỏ tới nó.
interface Product {
  readonly code: string;
  readonly name: string;
  readonly description: string;
  readonly image: string;
}

// Factory: mỗi mã sản phẩm chỉ tạo đúng một đối tượng, những lần sau trả lại đối tượng cũ.
class ProductCatalog {
  private made = new Map<string, Product>();

  get(code: string): Product {
    let product = this.made.get(code);
    if (!product) {
      const [name, description, image] = details(code);
      product = Object.freeze({ code, name, description, image });
      this.made.set(code, product);
    }
    return product;
  }

  get size(): number {
    return this.made.size;
  }
}

// Phần riêng của từng dòng: số lượng, cùng một tham chiếu tới sản phẩm dùng chung.
class OrderLine {
  product: Product;
  quantity: number;

  constructor(product: Product, quantity: number) {
    this.product = product;
    this.quantity = quantity;
  }

  label(): string {
    return `${this.product.name} x${this.quantity}`;
  }
}

const group = (n: number) => String(n).replace(/\B(?=(\d{3})+(?!\d))/g, ".");
const size = (bytes: number) => (bytes >= 1e6 ? `${(bytes / 1e6).toFixed(1)} MB` : `${(bytes / 1000).toFixed(1)} KB`).replace(".", ",");

function summary(title: string, rows: [string, number][]): OrderLine[] {
  const catalog = new ProductCatalog();
  const lines = rows.map(([code, quantity]) => new OrderLine(catalog.get(code), quantity));
  const memory = catalog.size * PRODUCT_BYTES + lines.length * (LINE_BYTES + REF_BYTES);
  console.log(title);
  console.log(`  đối tượng sản phẩm: ${group(catalog.size)}`);
  console.log(`  bộ nhớ ước tính: ${size(memory)}`);
  return lines;
}

Khi nào dùng

  • Chương trình phải giữ rất nhiều đối tượng cùng lúc, và bạn đã đo được bộ nhớ là vấn đề thật.
  • Phần lớn dữ liệu của các đối tượng ấy lặp lại và không đổi, như thông tin sản phẩm trong dòng đơn hàng, kiểu chữ trong trình soạn thảo, loại ô trong sơ đồ kho hay loại cây trên bản đồ game.
  • Phần riêng của mỗi đối tượng nhỏ và tách ra được, để có thể truyền vào lúc dùng.

Khi nào không nên

  • Khi số đối tượng chỉ vài nghìn, phần bộ nhớ tiết kiệm được rất nhỏ so với độ phức tạp thêm vào.
  • Khi phần tưởng là dùng chung lại hay bị sửa theo từng đối tượng, vì khi ấy không thể dùng chung một cách an toàn.
  • Khi chưa đo. Hãy đo bộ nhớ trước, vì nút thắt nhiều khi nằm ở chỗ khác.

Dễ nhầm với

  • Singleton chỉ cho đúng một thể hiện của cả lớp. Flyweight có nhiều thể hiện, mỗi giá trị khác nhau một thể hiện, và chúng được dùng chung theo giá trị.
  • Cache cũng giữ lại đối tượng để dùng lại, nhưng mục đích thường là tránh tính lại hay tải lại. Flyweight nhắm vào bộ nhớ và đòi đối tượng dùng chung phải bất biến.
  • Prototype sao chép một mẫu để có đối tượng mới độc lập. Flyweight làm điều ngược lại: không sao chép mà cùng trỏ tới một đối tượng.

Ghi chú khi triển khai

  • Đối tượng dùng chung phải bất biến. Python có @dataclass(frozen=True), TypeScript có Object.freeze đi kèm readonly. Nếu không, sửa ở một dòng là sửa luôn mọi dòng đang trỏ tới nó, đúng như bước cuối của hình.
  • Chuỗi giống nhau cũng có thể dùng chung theo cách này. Python tự gộp nhiều chuỗi ngắn và có sys.intern cho chuỗi bạn tự tạo, còn các engine JavaScript cũng gộp chuỗi literal. Đó là Flyweight ở mức ngôn ngữ.
  • Hãy để factory là nơi duy nhất tạo flyweight. Chỉ cần một chỗ khác gọi thẳng constructor là bản sao lại xuất hiện.
  • Khi nhiều thread cùng gọi factory, bảng tra cần được khoá, hoặc bạn tạo sẵn toàn bộ flyweight ngay từ đầu.
  • Con số trong bài chỉ là ước tính để thấy tỉ lệ. Muốn biết thật, hãy đo bằng tracemalloc trong Python hoặc heap snapshot trong Node.

Kiểm tra nhanh

Sáu câu hỏi ngắn để xem bạn đã nắm Flyweight tới đâu. Chọn một đáp án, lời giải thích sẽ hiện ngay bên dưới.

  1. Câu 1 Một trình soạn thảo lưu mỗi ký tự thành một đối tượng, kèm phông chữ, cỡ chữ và kiểu chữ. Một tài liệu dài có hai triệu ký tự nhưng chỉ dùng năm kiểu chữ. Cách nào giảm bộ nhớ lúc đang chạy nhiều nhất?

    Phần kiểu chữ lặp lại và không đổi nên gom được về năm flyweight, còn mỗi ký tự chỉ giữ phần riêng của nó. Singleton không hợp vì tài liệu cần năm kiểu chứ không phải một, còn nén khi lưu xuống đĩa chẳng giúp gì cho bộ nhớ lúc đang chạy.

  2. Câu 2 Ở bước đầu của hình, mười hai dòng chiếm 24,8 KB. Sang bước hai, cũng mười hai dòng ấy chỉ còn 6,9 KB. Phần chênh lệch đến từ đâu?

    Mười hai bản thông tin sản phẩm, mỗi bản 2.000 byte, rút xuống còn ba bản. Mỗi dòng vẫn giữ phần riêng 64 byte và có thêm 8 byte tham chiếu, nên tổng cộng là 6.864 byte, khoảng 6,9 KB.

  3. Câu 3 Trong code của bài, đâu là extrinsic state, tức phần riêng của từng dòng?

    Số lượng khác nhau ở mỗi dòng nên nằm trong OrderLine. Tên và ảnh giống hệt nhau ở mọi dòng cùng sản phẩm nên thuộc về flyweight Product, còn bảng tra là nơi factory cất các flyweight đã tạo.

  4. Câu 4 Vì sao Product được khai báo với frozen=True trong Python và Object.freeze trong TypeScript?

    Flyweight được nhiều dòng dùng chung, nên sửa nó cho một dòng cũng là sửa cho tất cả. Việc mỗi mã chỉ có một đối tượng là nhờ factory, còn bất biến là để không ai vô tình sửa đối tượng ấy.

  5. Câu 5 Flyweight khác Singleton ở điểm nào?

    Trong bài có ba thể hiện Product cho ba sản phẩm của lô nhỏ và 300 thể hiện cho cả báo cáo. Singleton không làm được vậy, vì nó chỉ cho đúng một thể hiện.

  6. Câu 6 Trường hợp nào không đáng dùng Flyweight?

    Khi đối tượng ít và hầu như không có dữ liệu lặp lại thì chẳng có gì để dùng chung, nên Flyweight chỉ thêm factory và độ phức tạp. Ba trường hợp còn lại đều có rất nhiều đối tượng mang chung một phần nặng.
